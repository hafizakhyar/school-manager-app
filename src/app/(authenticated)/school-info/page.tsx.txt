"use client";

import React, { useState, useEffect } from "react";
import { db } from "@/lib/firebase";
import { useAuth } from "@/context/auth-context";
import { 
  collection, 
  getDocs, 
  addDoc, 
  deleteDoc, 
  doc, 
  Timestamp, 
  query, 
  orderBy 
} from "firebase/firestore";
import { Megaphone, Plus, Trash2, Calendar, Tag, X, BellRing } from "lucide-react";

interface Announcement {
  id: string;
  title: string;
  content: string;
  category: string;
  createdAt: any;
}

export default function SchoolInfoPage() {
  const { role } = useAuth();
  const [announcements, setAnnouncements] = useState<Announcement[]>([]);
  const [loading, setLoading] = useState(true);

  // Modal State (Admin only)
  const [isOpen, setIsOpen] = useState(false);
  const [title, setTitle] = useState("");
  const [content, setContent] = useState("");
  const [category, setCategory] = useState("Pengumuman");

  const fetchAnnouncements = async () => {
    setLoading(true);
    try {
      const q = query(collection(db, "schoolInfo"), orderBy("createdAt", "desc"));
      const snapshot = await getDocs(q);
      const list: Announcement[] = [];
      snapshot.forEach(d => {
        list.push({ id: d.id, ...d.data() } as Announcement);
      });
      setAnnouncements(list);
    } catch (err) {
      console.error("Gagal memuat informasi sekolah:", err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchAnnouncements();
  }, []);

  const handleSave = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!title || !content) {
      alert("Judul dan isi pengumuman wajib diisi!");
      return;
    }

    try {
      await addDoc(collection(db, "schoolInfo"), {
        title,
        content,
        category,
        createdAt: Timestamp.now()
      });
      setTitle("");
      setContent("");
      setCategory("Pengumuman");
      setIsOpen(false);
      fetchAnnouncements();
    } catch (error) {
      console.error("Gagal menyimpan pengumuman:", error);
      alert("Gagal mempublikasikan informasi.");
    }
  };

  const handleDelete = async (id: string) => {
    if (!confirm("Yakin ingin menghapus informasi ini?")) return;
    try {
      await deleteDoc(doc(db, "schoolInfo", id));
      fetchAnnouncements();
    } catch (error) {
      console.error("Gagal menghapus:", error);
      alert("Gagal menghapus data.");
    }
  };

  const getCategoryBadge = (cat: string) => {
    switch (cat) {
      case "Agenda": return "bg-amber-500/10 text-amber-400 border-amber-500/20";
      case "Penting": return "bg-red-500/10 text-red-400 border-red-500/20";
      default: return "bg-indigo-500/10 text-indigo-400 border-indigo-500/20";
    }
  };

  return (
    <div className="space-y-8">
      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h1 className="text-3xl font-extrabold text-white tracking-tight">Informasi & Agenda Sekolah</h1>
          <p className="text-sm text-slate-400 mt-1">Papan pengumuman resmi, berita, dan agenda kegiatan sekolah terbaru.</p>
        </div>

        {role === "admin" && (
          <button 
            onClick={() => setIsOpen(true)} 
            className="flex items-center justify-center gap-2 rounded-lg bg-indigo-600 px-4 py-2.5 text-sm font-semibold text-white hover:bg-indigo-500 cursor-pointer transition-colors"
          >
            <Plus className="h-4.5 w-4.5" />
            <span>Buat Pengumuman Baru</span>
          </button>
        )}
      </div>

      {loading ? (
        <div className="space-y-4">
          {[1, 2, 3].map(n => <div key={n} className="h-28 w-full animate-pulse rounded-xl bg-slate-900" />)}
        </div>
      ) : (
        <div className="grid grid-cols-1 gap-6">
          {announcements.length === 0 ? (
            <div className="rounded-2xl border border-slate-900 bg-slate-900/40 p-12 text-center">
              <BellRing className="mx-auto h-10 w-10 text-slate-600 mb-3" />
              <p className="text-sm text-slate-400 font-medium">Belum ada pengumuman atau agenda sekolah saat ini.</p>
            </div>
          ) : (
            announcements.map((item) => (
              <div key={item.id} className="rounded-2xl border border-slate-900 bg-slate-900/40 p-6 backdrop-blur-sm relative group hover:border-slate-800 transition-all">
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 mb-4">
                  <div className="flex items-center gap-3">
                    <span className={`rounded-full px-3 py-1 text-xs font-semibold uppercase tracking-wider border ${getCategoryBadge(item.category)}`}>
                      {item.category}
                    </span>
                    <div className="flex items-center gap-1.5 text-xs text-slate-400">
                      <Calendar className="h-3.5 w-3.5 text-slate-500" />
                      <span>
                        {item.createdAt?.toDate ? item.createdAt.toDate().toLocaleDateString("id-ID", { day: 'numeric', month: 'long', year: 'numeric' }) : "Baru saja"}
                      </span>
                    </div>
                  </div>

                  {role === "admin" && (
                    <button 
                      onClick={() => handleDelete(item.id)} 
                      className="absolute top-6 right-6 rounded-lg p-2 text-slate-500 hover:text-red-400 hover:bg-red-500/10 transition-colors cursor-pointer"
                      title="Hapus Pengumuman"
                    >
                      <Trash2 className="h-4 w-4" />
                    </button>
                  )}
                </div>

                <h3 className="text-lg font-bold text-white mb-2 pr-8">{item.title}</h3>
                <p className="text-sm text-slate-300 whitespace-pre-line leading-relaxed">{item.content}</p>
              </div>
            ))
          )}
        </div>
      )}

      {/* Modal Tambah Pengumuman (Admin Only) */}
      {isOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
          <div className="fixed inset-0 bg-slate-950/60 backdrop-blur-sm" onClick={() => setIsOpen(false)} />
          <div className="relative w-full max-w-lg rounded-2xl border border-slate-800 bg-slate-900 p-6 shadow-2xl">
            <div className="flex items-center justify-between border-b border-slate-800 pb-4 mb-4">
              <h2 className="text-lg font-bold text-white">Buat Informasi / Agenda Baru</h2>
              <button onClick={() => setIsOpen(false)} className="text-slate-400 hover:text-white cursor-pointer"><X className="h-5 w-5" /></button>
            </div>
            <form onSubmit={handleSave} className="space-y-4">
              <div>
                <label className="block text-xs font-bold uppercase text-slate-400 mb-1">Judul Pengumuman / Agenda</label>
                <input 
                  type="text" 
                  value={title} 
                  onChange={e => setTitle(e.target.value)} 
                  placeholder="Contoh: Jadwal Ujian Tengah Semester Genap"
                  className="w-full rounded-lg border border-slate-800 bg-slate-950 px-3 py-2 text-sm text-slate-200 outline-none focus:border-indigo-500" 
                />
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-slate-400 mb-1">Kategori</label>
                <select 
                  value={category} 
                  onChange={e => setCategory(e.target.value)}
                  className="w-full rounded-lg border border-slate-800 bg-slate-950 px-3 py-2 text-sm text-slate-200 outline-none focus:border-indigo-500"
                >
                  <option value="Pengumuman">Pengumuman Umum</option>
                  <option value="Agenda">Agenda Kegiatan</option>
                  <option value="Penting">Penting / Darurat</option>
                </select>
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-slate-400 mb-1">Isi Pesan / Keterangan Lengkap</label>
                <textarea 
                  rows={5}
                  value={content} 
                  onChange={e => setContent(e.target.value)} 
                  placeholder="Tuliskan detail informasi atau agenda di sini..."
                  className="w-full rounded-lg border border-slate-800 bg-slate-950 px-3 py-2 text-sm text-slate-200 outline-none focus:border-indigo-500 resize-none" 
                />
              </div>

              <div className="flex justify-end gap-3 pt-4 border-t border-slate-800">
                <button type="button" onClick={() => setIsOpen(false)} className="px-4 py-2 text-sm text-slate-400 cursor-pointer">Batal</button>
                <button type="submit" className="rounded-lg bg-indigo-600 px-4 py-2 text-sm font-semibold text-white hover:bg-indigo-500 cursor-pointer">Publikasikan</button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
